def rat_num_row():
    f=open('Danish.txt','w+')
    f.write('rat num row\n-------------------------\n')
    for i in range(1,6):
        for j in range(0,i):
            f.write(str(i))
        f.write('\n')
    f.close()
rat_num_row()
         
 
def rat_num_col():
    f=open('Danish.txt','a+')
    f.write('\n')

    for i in range(1,6):
        for j in range(0,i):
            f.write(str(j))
        f.write('\n')
    f.close()
rat_num_col()

def rat_name_row():
    f=open('Danish.txt','a+')
    f.write('\n')
    f.write('rat name row\n-------------------------\n')
    name='danish'
    for i in range(len(name)):
        for j in range(i+1):
            f.write(name[i]+' ')
        f.write('\n')
    f.close()
rat_name_row()
