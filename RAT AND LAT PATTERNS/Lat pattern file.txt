def lat_num_row():
    f=open('DanishILat.txt','w+')
    f.write('\n')
    f.write('lat num row\n-------------------------\n')
    for i in range(6,0,-1):
        for j in range(6,i,-1):
            f.write(' ')
        for k in range(0,i):
            f.write(str(i))
        f.write('\n')
    f.close()
lat_num_row()

def lat_name_row():
    f=open('DanishILat.txt','a+')
    f.write('\n')
    f.write('lat name row\n-------------------------\n')
    name='danish'
    for i in range(len(name)):
        for j in range(len(name),i,-1):
            f.write(' ')
        for k in range(i+1):
            f.write(name[i]+'')
        f.write('\n')
    f.close()
lat_name_row()
