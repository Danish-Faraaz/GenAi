def ilat_num_row():
    f=open('DanishILat.txt','w+')
    f.write('\n')
    f.write('ilat num row\n-------------------------\n')
    for i in range(6,0,-1):
        for j in range(6,i,-1):
            f.write(' ')
        for k in range(0,i):
            f.write(str(i))
        f.write('\n')
    f.close()
ilat_num_row()
