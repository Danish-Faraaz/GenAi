def irat_num_row():
    f=open('Danish.txt','w+')
    f.write('irat num row\n-------------------------\n')
    for i in range(5,0,-1):
        for j in range(1,i):
            f.write('')
        for k in range(0,i):
            f.write(str(i)+' ')
        f.write('\n')
    f.close()
irat_num_row()

def irat_name_row():
