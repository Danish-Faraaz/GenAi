def prt_num_row():
    f=open('Prt pattern file.txt','w+')
    f.write('\n')
    f.write('prt num row\n-------------------------\n')
    for i in range(1,6):
        for j in range(6,i,-1):
            f.write(' ')
        for k in range(0,i):
            f.write(str(i)+' ')
        f.write('\n')
    f.close()
prt_num_row()
